Supabase テーブル作成SQL

Supabaseの管理画面左メニュー 「SQL Editor」 ＞ 「New query」 に以下のコードを貼り付けて 「Run」 を押してください。大学note（UniPass）の既存テーブルとも互換対応しています。 

-- 1. 時間割・講義テーブル
create table if not exists classes (
  id text primary key,
  name text not null,
  day integer,
  period integer,
  teacher text,
  room text,
  instructor text,
  location text,
  year text,
  semester text,
  url_template text,
  type text default 'class',
  memo text,
  color text,
  materials jsonb default '[]'::jsonb,
  notes jsonb default '[]'::jsonb,
  updated_at timestamp with time zone default timezone('utc'::text, now())
);

-- 既存テーブルがある場合の拡張（安全策）
alter table classes add column if not exists teacher text;
alter table classes add column if not exists room text;
alter table classes add column if not exists instructor text;
alter table classes add column if not exists location text;
alter table classes add column if not exists year text;
alter table classes add column if not exists semester text;
alter table classes add column if not exists url_template text;
alter table classes add column if not exists type text default 'class';
alter table classes add column if not exists memo text;

-- 2. リアルタイム同期の有効化
alter publication supabase_realtime add table classes;
